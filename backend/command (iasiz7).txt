$adb = Join-Path $env:LOCALAPPDATA 'Android\Sdk\platform-tools\adb.exe'; $fin = (Get-Date).AddSeconds(120); $prev = ''; $changes = 0; $pretDepuis = $null; while ((Get-Date) -lt $fin) { $p = "$(& $adb shell pidof system_server 2>$null)".Trim(); $svc = "$(& $adb shell service check package 2>$null)".Trim(); $anim = "$(& $adb shell getprop init.svc.bootanim 2>$null)".Trim(); if ($p -ne $prev) { "$(Get-Date -Format 'HH:mm:ss') system_server=$p package=$svc anim=$anim"; if ($prev) { $changes++ }; $prev = $p }; if ($svc -match 'found' -and $svc -notmatch 'not found' -and $anim -eq 'stopped') { if (-not $pretDepuis) { $pretDepuis = Get-Date } elseif (((Get-Date) - $pretDepuis).TotalSeconds -ge 20) { break } } else { $pretDepuis = $null }; Start-Sleep -Seconds 3 }; "redemarrages observes: $changes"; "stable: $([bool]$pretDepuis)"; "package: $("$(& $adb shell service check package 2>$null)".Trim())"; "anim: $("$(& $adb shell getprop init.svc.bootanim 2>$null)".Trim())"; 'raison du dernier redemarrage:'; & $adb logcat -d -b system,crash -t 400 2>$null | Select-String -Pattern 'FATAL EXCEPTION IN SYSTEM|\*\*\* FATAL|system_server|Watchdog|Out of memory|lowmemorykiller' | ForEach-Object { $l=$_.Line; if ($l.Length -gt 190) { $l.Substring(0,190) } else { $l } } | Select-Object -Last 8
